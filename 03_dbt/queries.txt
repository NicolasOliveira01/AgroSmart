create database exists AGROSMART;
create schema exists AGROSMART.RAW;

create file format AGROSMART.RAW.FFT_JSONL
    type = json
    compression = auto;

create stage AGROSMART.RAW.STG_LEITURAS
    file_format = AGROSMART.RAW.FFT_JSONL;

create table if not exists AGROSMART.RAW.LEITURAS_SENSORES (
    _arquivo        string        comment 'Nome do arquivo de origem no stage',
    _linha_arquivo  number        comment 'Número da linha dentro do arquivo',
    _carregado_em   timestamp_ntz comment 'Momento da carga no Snowflake',
    _doc            variant       comment 'Leitura completa do sensor em JSON'
);
